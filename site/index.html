import { setTimeout as sleep } from "node:timers/promises";
import type { Delivery, Endpoint } from "../types";
import { metrics, tracer } from "../telemetry";

const MAX_ATTEMPTS = 8;
const BASE_DELAY_MS = 250;
const CEILING_MS = 60_000;

/** Equal jitter, capped: 250 ms, 500 ms, 1 s … 60 s. */
export function backoff(attempt: number, jitter = Math.random()): number {
  const exp = Math.min(CEILING_MS, BASE_DELAY_MS * 2 ** attempt);
  return Math.round(exp / 2 + (exp / 2) * jitter);
}

export async function deliver(d: Delivery, ep: Endpoint): Promise<boolean> {
  for (let attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
    const res = await fetch(ep.url, {
      method: "POST",
      headers: { "content-type": "application/json", "x-relay-id": d.id },
      body: JSON.stringify(d.payload),
      signal: AbortSignal.timeout(10_000),
    });
    if (res.ok) {
      metrics.increment("delivery.ok", { endpoint: ep.id });
      return true;
    }
    if (res.status >= 400 && res.status < 500 && res.status !== 429) break;

    const retryAfter = Number(res.headers.get("retry-after"));
    const wait = Number.isFinite(retryAfter) && retryAfter > 0
      ? Math.min(CEILING_MS, retryAfter * 1_000)
      : backoff(attempt);
    await sleep(wait);
  }
  metrics.increment("delivery.failed", { endpoint: ep.id });
  return false;
}

import { describe, expect, it } from "vitest";
import { backoff } from "../src/queue/retry";

describe("backoff", () => {
  it("is half the window with zero jitter", () => {
    expect(backoff(0, 0)).toBe(125);
    expect(backoff(3, 0)).toBe(1_000);
  });

  it("never exceeds CEILING_MS", () => {
    for (let n = 0; n < 32; n++) {
      expect(backoff(n, 1)).toBeLessThanOrEqual(60_000);
    }
  });

  it("doubles until the ceiling", () => {
    const delays = [0, 1, 2, 3, 4].map((n) => backoff(n, 1));
    expect(delays).toEqual([250, 500, 1_000, 2_000, 4_000]);
  });
});

Columns that line up by themselves.

Every figure takes the same advance, so psql's padding alone aligns money, ratios and dates. Zero, O, one, I and l stay distinct inside the IDs.

SELECT a.account_id,
       a.plan,
       count(d.id)                         AS deliveries_30d,
       round(avg((d.status >= 400)::int), 4) AS failed_ratio,
       a.credit_balance_usd,
       max(p.paid_at)::date                AS last_payment_at,
       a.is_suspended
  FROM accounts a
  LEFT JOIN deliveries d
         ON d.account_id = a.account_id
        AND d.created_at > now() - interval '30 days'
  LEFT JOIN payments p ON p.account_id = a.account_id
 GROUP BY a.account_id
 ORDER BY failed_ratio DESC NULLS LAST, deliveries_30d DESC;

Reference / queue

backoff()

Returns the delay, in milliseconds, before attempt n is retried. Delays double from BASE_DELAY_MS and stop at CEILING_MS; jitter spreads retries from many workers so they do not reach an endpoint in lockstep. 

function backoff(attempt: number, jitter?: number): number

Parameters

NameTypeDefaultDescription
attemptnumber—Zero-based attempt index.
jitternumberMath.random()In [0, 1). Zero gives the lower bound.

Returns

number, in milliseconds, between half the window and the full window, never above 60_000.

Note A 429 or 503 carrying Retry-After overrides the computed delay. See Delivery semantics.

backoff(0, 0);   // 125
backoff(3, 0);   // 1_000
backoff(20, 1);  // 60_000, capped

NFNerd Font geometry

NF or NFM?

Each Nerd Font archive ships two patched variants with the same icons. They differ only in how icons sit in the cell.

NF Ioskeley Mono Nerd Font
Ordinary Nerd Font geometry. Larger icons stay intact and may extend a little past one cell, which looks right in most prompts. A good default.
NFM Ioskeley Mono Nerd Font Mono
Strict single-cell geometry: every glyph, icon included, stays inside one cell. What Fontconfig and some font pickers require of a true monospace face.

Applies to 2.2.0. The currently published v2.1.0 archives do not contain this split, and ship neither variant. If you are on v2.1.0, ignore this section until 2.2.0 is released. OriginsDesign evolution

Where it started

Berkeley Mono / TX-02 inspired the first experiment, and Iosevka made it possible to run. Both are history rather than specification: a hundred-odd builds and repeated tuning moved Ioskeley toward a design of its own. The studies below are kept as a record of that process. They are comparison plates, deliberately monochrome, and they are not a claim of equivalence.

Install & use

Manually

Unzip, choose Normal or SemiCondensed, then Hinted for standard-DPI Windows and Linux or Unhinted for macOS and HiDPI.

Windows: right-click → Install. macOS: Font Book. Linux: copy to ~/.local/share/fonts/, run fc-cache -fv.
